   174→   174→    function tick() {

   184→   184→      if (keys.has("s") || keys.has("arrowdown")) dy += 1;
   185→   185→      if (keys.has("a") || keys.has("arrowleft")) dx -= 1;
   186→   186→      if (keys.has("d") || keys.has("arrowright")) dx += 1;
   187→   187→

   191→   191→        const mag = Math.sqrt(j.dx * j.dx + j.dy * j.dy);
   192→   192→        if (mag > 10) {
   193→   193→          dx = j.dx / mag;
   194→   194→          dy = j.dy / mag;
   195→   195→        }
   196→   196→      }
   197→   197→
   198→   198→      const speed = 3.5;
   199→   199→      const moving = (dx !== 0 || dy !== 0);
   200→   200→      if (moving) {
   201→   201→        const norm = Math.sqrt(dx * dx + dy * dy) || 1;
   202→   202→        const ndx = dx / norm;
   203→   203→        const ndy = dy / norm;

   205→   205→        playerPos.current.y = Math.max(60, Math.min(area.height - 20, playerPos.current.y + ndy * speed));

   208→   208→          playerPos.current.direction = ndx > 0 ? "right" : "left";
   209→   209→        } else {
   210→   210→          playerPos.current.direction = ndy > 0 ? "down" : "up";
   211→   211→        }
   212→   212→        playerPos.current.isMoving = true;
   213→   213→      } else {
   214→   214→        playerPos.current.isMoving = false;
   215→   215→      }
   216→   216→
   217→   217→      // === Sync to server (~30Hz) ===

   220→   220→        moveTo(playerPos.current.x, playerPos.current.y, playerPos.current.direction);

   238→   238→      ctx!.lineWidth = 1;
   239→   239→      const gridSize = 60;
   240→   240→      const startX = Math.floor(camX / gridSize) * gridSize;
   241→   241→      const startY = Math.floor(camY / gridSize) * gridSize;

   243→   243→        ctx!.beginPath();
   244→   244→        ctx!.moveTo(x, camY);
   245→   245→        ctx!.lineTo(x, camY + h);
   246→   246→        ctx!.stroke();
   247→   247→      }
   248→   248→      for (let y = startY; y < camY + h; y += gridSize) {
   249→   249→        ctx!.beginPath();
   250→   250→        ctx!.moveTo(camX, y);
   251→   251→        ctx!.lineTo(camX + w, y);
   252→   252→        ctx!.stroke();
   253→   253→      }

   259→   259→

   397→   397→        }

   400→   400→      // === Other players (remote) ===
   401→   401→      for (const p of players) {
