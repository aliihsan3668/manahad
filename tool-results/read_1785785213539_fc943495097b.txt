   126→export function ModeratorView() {
   127→  const user = useAppStore((s) => s.user);
   128→  const setView = useAppStore((s) => s.setView);
   129→  const [data, setData] = useState<ModData | null>(null);
   130→  const [loading, setLoading] = useState(true);
   131→  const [actionTarget, setActionTarget] = useState<{ userId: string; displayName: string } | null>(null);
   132→  const [busyId, setBusyId] = useState<string | null>(null);
   133→
   134→  const load = useCallback(async () => {
   135→    setLoading(true);
   136→    try {
   137→      const res = await fetch("/api/moderator/dashboard");
   138→      const d = await res.json();
   139→      if (res.ok) setData(d);
   140→      else toast.error(d.error ?? "Could not load dashboard");
   141→    } catch {
   142→      toast.error("Network error");
   143→    } finally {
   144→      setLoading(false);
   145→    }
   146→  }, []);
   147→
   148→  useEffect(() => { load(); }, [load]);
   149→
   150→  async function takeAction(params: {
   151→    actionType: string; userId: string; reason: string; durationMinutes: number;
   152→  }) {
   153→    setBusyId(params.userId);
   154→    try {
   155→      const res = await fetch("/api/moderator/action", {
   156→        method: "POST",
   157→        headers: { "Content-Type": "application/json" },
   158→        body: JSON.stringify(params),
   159→      });
   160→      const d = await res.json();
   161→      if (!res.ok) throw new Error(d.error ?? "Action failed");
   162→      toast.success(`✅ ${params.actionType} applied to user`);
   163→      setActionTarget(null);
   164→      await load();
   165→    } catch (e) {
   166→      toast.error(e instanceof Error ? e.message : "Action failed");
   167→    } finally {
   168→      setBusyId(null);
   169→    }
   170→  }
   171→
   172→  async function reviewAppeal(appealId: string, status: "APPROVED" | "DENIED") {
   173→    setBusyId(appealId);
   174→    try {
   175→      const res = await fetch("/api/moderator/appeal", {
   176→        method: "POST",
   177→        headers: { "Content-Type": "application/json" },
   178→        body: JSON.stringify({ appealId, status }),
   179→      });
   180→      const d = await res.json();
   181→      if (!res.ok) throw new Error(d.error ?? "Appeal review failed");
   182→      toast.success(`Appeal ${status.toLowerCase()}`);
   183→      await load();
   184→    } catch (e) {
   185→      toast.error(e instanceof Error ? e.message : "Could not review appeal");
   186→    } finally {
   187→      setBusyId(null);
   188→    }
   189→  }
   190→
   191→  if (user?.role !== "MODERATOR" && user?.role !== "ADMIN") {
   192→    return (
   193→      <div className="min-h-screen flex items-center justify-center p-4">
   194→        <Card className="max-w-md w-full border-rose-300">
   195→          <CardHeader>
   196→            <CardTitle className="flex items-center gap-2 text-rose-700">
   197→              <AlertTriangle className="w-5 h-5" /> Access Denied
   198→            </CardTitle>
   199→            <CardDescription>
   200→              Moderator or Admin privileges are required.
   201→            </CardDescription>
   202→          </CardHeader>
   203→          <CardContent>
   204→            <Button onClick={() => setView("auth")} className="w-full">
   205→              <ArrowLeft className="w-4 h-4 mr-1" /> Back to Login
   206→            </Button>
   207→          </CardContent>
   208→        </Card>
   209→      </div>
   210→    );
   211→  }
   212→
   213→  return (
   214→    <div className="min-h-screen p-4 md:p-6 max-w-7xl mx-auto pb-12">
   215→      {/* Header */}
   216→      <div className="flex items-center justify-between mb-6 flex-wrap gap-3">
   217→        <div>
   218→          <h1 className="text-2xl md:text-3xl font-bold flex items-center gap-2">
   219→            <ShieldAlert className="w-7 h-7 text-rose-600" /> Moderator Dashboard
   220→          </h1>
   221→          <p className="text-sm text-muted-foreground mt-1">
   222→            Review reports, manage penalties, and resolve appeals.
   223→          </p>
   224→        </div>
   225→        <div className="flex gap-2">
   226→          <Button variant="outline" size="sm" onClick={load}>
   227→            <RefreshCw className="w-4 h-4 mr-1" /> Refresh
   228→          </Button>
   229→          <Button variant="ghost" size="sm" onClick={() => setView("auth")}>
   230→            <ArrowLeft className="w-4 h-4 mr-1" /> Logout
   231→          </Button>
   232→        </div>
   233→      </div>
   234→
   235→      {/* Summary cards */}
   236→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3 mb-6">
   237→        <SummaryCard icon={Flag} label="Open Reports" value={data?.openReports.length ?? 0} color="text-rose-600" />
   238→        <SummaryCard icon={MessageSquareWarning} label="Flagged Msgs" value={data?.recentFlaggedMessages.length ?? 0} color="text-amber-600" />
   239→        <SummaryCard icon={Gavel} label="Active Penalties" value={data?.activeMutesSuspensions.length ?? 0} color="text-purple-600" />
   240→        <SummaryCard icon={Scale} label="Open Appeals" value={data?.openAppeals.length ?? 0} color="text-sky-600" />
   241→      </div>
   242→
   243→      {loading || !data ? (
   244→        <div className="space-y-3">
   245→          <Skeleton className="h-12" />
   246→          <Skeleton className="h-64" />
   247→        </div>
   248→      ) : (
   249→        <Tabs defaultValue="reports">
   250→          <TabsList className="grid w-full grid-cols-2 md:grid-cols-4 mb-4">
   251→            <TabsTrigger value="reports"><Flag className="w-4 h-4 mr-1" /> Reports</TabsTrigger>
   252→            <TabsTrigger value="flagged"><MessageSquareWarning className="w-4 h-4 mr-1" /> Flagged</TabsTrigger>
   253→            <TabsTrigger value="penalties"><Gavel className="w-4 h-4 mr-1" /> Penalties</TabsTrigger>
   254→            <TabsTrigger value="appeals"><Scale className="w-4 h-4 mr-1" /> Appeals</TabsTrigger>
   255→          </TabsList>
   256→
   257→          {/* ============== REPORTS ============== */}
   258→          <TabsContent value="reports">
   259→            <Card>
   260→              <CardHeader>
   261→                <CardTitle className="text-lg">Open Reports</CardTitle>
   262→                <CardDescription>User-flagged chat messages awaiting review.</CardDescription>
   263→              </CardHeader>
   264→              <CardContent>
   265→                {data.openReports.length === 0 ? (
   266→                  <EmptyState text="No open reports. 🎉" />
   267→                ) : (
   268→                  <Table>
   269→                    <TableHeader>
   270→                      <TableRow>
   271→                        <TableHead>Reported User</TableHead>
   272→                        <TableHead>Message</TableHead>
   273→                        <TableHead className="hidden md:table-cell">Reason</TableHead>
   274→                        <TableHead>Status</TableHead>
   275→                        <TableHead>Created</TableHead>
   276→                        <TableHead className="text-right">Action</TableHead>
   277→                      </TableRow>
   278→                    </TableHeader>
   279→                    <TableBody>
   280→                      {data.openReports.map((r) => (
   281→                        <TableRow key={r.id}>
   282→                          <TableCell>
   283→                            <div className="flex items-center gap-2">
   284→                              <UserCircle className="w-4 h-4 text-muted-foreground" />
   285→                              <div>
   286→                                <p className="font-medium text-sm">{r.reportedUser.displayName}</p>
   287→                                <p className="text-[10px] text-muted-foreground">@{r.reportedUser.username}</p>
   288→                              </div>
   289→                            </div>
   290→                          </TableCell>
   291→                          <TableCell className="max-w-[200px]">
   292→                            <p className="text-xs line-clamp-2">{r.message.displayedContent || r.message.rawContent}</p>
   293→                          </TableCell>
   294→                          <TableCell className="hidden md:table-cell text-xs">{r.reason}</TableCell>
   295→                          <TableCell>
   296→                            <Badge className={`text-[10px] ${STATUS_COLOR[r.status] ?? STATUS_COLOR.OPEN}`}>{r.status}</Badge>
   297→                          </TableCell>
   298→                          <TableCell className="text-xs text-muted-foreground">
   299→                            {new Date(r.createdAt).toLocaleDateString()}
   300→                          </TableCell>
   301→                          <TableCell className="text-right">
   302→                            <TakeActionDialog
   303→                              target={{ userId: r.reportedUser.id, displayName: r.reportedUser.displayName }}
   304→                              onConfirm={takeAction}
   305→                              busy={busyId === r.reportedUser.id}
   306→                            />
   307→                          </TableCell>
   308→                        </TableRow>
   309→                      ))}
   310→                    </TableBody>
   311→                  </Table>
   312→                )}
   313→              </CardContent>
   314→            </Card>
   315→          </TabsContent>
   316→
   317→          {/* ============== FLAGGED MESSAGES ============== */}
   318→          <TabsContent value="flagged">
   319→            <Card>
   320→              <CardHeader>
   321→                <CardTitle className="text-lg">Flagged Messages</CardTitle>
   322→                <CardDescription>Recent messages blocked, escalated, or rewritten by AI.</CardDescription>
   323→              </CardHeader>
   324→              <CardContent>
   325→                {data.recentFlaggedMessages.length === 0 ? (
   326→                  <EmptyState text="No flagged messages." />
   327→                ) : (
   328→                  <Table>
   329→                    <TableHeader>
   330→                      <TableRow>
   331→                        <TableHead>User</TableHead>
   332→                        <TableHead>Raw Content</TableHead>
   333→                        <TableHead>Displayed</TableHead>
   334→                        <TableHead>Status</TableHead>
   335→                        <TableHead>Reason</TableHead>
   336→                        <TableHead>Date</TableHead>
   337→                      </TableRow>
   338→                    </TableHeader>
   339→                    <TableBody>
   340→                      {data.recentFlaggedMessages.map((m) => (
   341→                        <TableRow key={m.id}>
   342→                          <TableCell>
   343→                            <p className="font-medium text-sm">{m.user.displayName}</p>
   344→                            <p className="text-[10px] text-muted-foreground">@{m.user.username}</p>
   345→                          </TableCell>
   346→                          <TableCell className="max-w-[180px] text-xs">
   347→                            <p className="line-clamp-2 text-rose-700 dark:text-rose-400">{m.rawContent}</p>
   348→                          </TableCell>
   349→                          <TableCell className="max-w-[180px] text-xs">
   350→                            <p className="line-clamp-2">{m.displayedContent || "—"}</p>
   351→                          </TableCell>
   352→                          <TableCell>
   353→                            <Badge className={`text-[10px] ${STATUS_COLOR[m.moderationStatus] ?? STATUS_COLOR.PENDING}`}>
   354→                              {m.moderationStatus}
   355→                            </Badge>
   356→                          </TableCell>
   357→                          <TableCell className="text-xs text-muted-foreground line-clamp-1">
   358→                            {m.moderationReason ?? "—"}
   359→                          </TableCell>
   360→                          <TableCell className="text-xs text-muted-foreground">
   361→                            {new Date(m.createdAt).toLocaleDateString()}
   362→                          </TableCell>
   363→                        </TableRow>
   364→                      ))}
   365→                    </TableBody>
   366→                  </Table>
   367→                )}
   368→              </CardContent>
   369→            </Card>
   370→          </TabsContent>
   371→
   372→          {/* ============== ACTIVE PENALTIES ============== */}
   373→          <TabsContent value="penalties">
   374→            <Card>
   375→              <CardHeader>
   376→                <CardTitle className="text-lg">Active Penalties</CardTitle>
   377→                <CardDescription>Currently enforced mutes, suspensions, and bans.</CardDescription>
   378→              </CardHeader>
   379→              <CardContent>
   380→                {data.activeMutesSuspensions.length === 0 ? (
   381→                  <EmptyState text="No active penalties. 🎉" />
   382→                ) : (
   383→                  <Table>
   384→                    <TableHeader>
   385→                      <TableRow>
   386→                        <TableHead>User</TableHead>
   387→                        <TableHead>Action</TableHead>
   388→                        <TableHead>Reason</TableHead>
   389→                        <TableHead>Expires</TableHead>
   390→                        <TableHead>Created</TableHead>
   391→                        <TableHead className="text-right">Action</TableHead>
   392→                      </TableRow>
   393→                    </TableHeader>
   394→                    <TableBody>
   395→                      {data.activeMutesSuspensions.map((a) => (
   396→                        <TableRow key={a.id}>
   397→                          <TableCell>
   398→                            <p className="font-medium text-sm">{a.user.displayName}</p>
   399→                            <p className="text-[10px] text-muted-foreground">@{a.user.username}</p>
   400→                          </TableCell>
   401→                          <TableCell>
   402→                            <Badge variant="outline" className={`text-[10px] ${ACTION_COLOR[a.actionType] ?? ""}`}>
   403→                              {a.actionType}
   404→                            </Badge>
   405→                          </TableCell>
   406→                          <TableCell className="text-xs line-clamp-1 max-w-[180px]">{a.reason}</TableCell>
   407→                          <TableCell className="text-xs text-muted-foreground">
   408→                            {a.expiresAt ? new Date(a.expiresAt).toLocaleString() : "Permanent"}
   409→                          </TableCell>
   410→                          <TableCell className="text-xs text-muted-foreground">
   411→                            {new Date(a.createdAt).toLocaleDateString()}
   412→                          </TableCell>
   413→                          <TableCell className="text-right">
   414→                            <Button
   415→                              size="sm"
   416→                              variant="outline"
   417→                              disabled={busyId === a.id}
   418→                              onClick={async () => {
   419→                                const liftAction = a.actionType === "MUTE" ? "UNMUTE" : a.actionType === "SUSPEND" ? "UNSUSPEND" : null;
   420→                                if (!liftAction) {
   421→                                  toast.error("Cannot lift a ban via this button — use appeals process.");
   422→                                  return;
   423→                                }
   424→                                setBusyId(a.id);
   425→                                try {
   426→                                  const res = await fetch("/api/moderator/action", {
   427→                                    method: "POST",
   428→                                    headers: { "Content-Type": "application/json" },
   429→                                    body: JSON.stringify({
   430→                                      actionType: liftAction,
   431→                                      userId: a.user.id,
   432→                                      reason: `Lifted by moderator (${user?.username ?? "unknown"})`,
   433→                                    }),
   434→                                  });
   435→                                  const d = await res.json();
   436→                                  if (!res.ok) throw new Error(d.error ?? "Lift failed");
   437→                                  toast.success(`${liftAction} applied`);
   438→                                  await load();
   439→                                } catch (e) {
   440→                                  toast.error(e instanceof Error ? e.message : "Could not lift penalty");
   441→                                } finally {
   442→                                  setBusyId(null);
   443→                                }
   444→                              }}
   445→                            >
   446→                              {busyId === a.id ? <Loader2 className="w-3 h-3 mr-1 animate-spin" /> : <Check className="w-3 h-3 mr-1" />}
   447→                              Lift
   448→                            </Button>
   449→                          </TableCell>
   450→                        </TableRow>
   451→                      ))}
   452→                    </TableBody>
   453→                  </Table>
   454→                )}
   455→              </CardContent>
   456→            </Card>
   457→          </TabsContent>
   458→
   459→          {/* ============== APPEALS ============== */}
   460→          <TabsContent value="appeals">
   461→            <Card>
   462→              <CardHeader>
   463→                <CardTitle className="text-lg">Open Appeals</CardTitle>
   464→                <CardDescription>Users contesting moderation actions.</CardDescription>
   465→              </CardHeader>
   466→              <CardContent>
   467→                {data.openAppeals.length === 0 ? (
   468→                  <EmptyState text="No open appeals." />
   469→                ) : (
   470→                  <div className="space-y-3">
   471→                    {data.openAppeals.map((a) => (
   472→                      <motion.div
   473→                        key={a.id}
   474→                        initial={{ opacity: 0, y: 4 }}
   475→                        animate={{ opacity: 1, y: 0 }}
   476→                        className="rounded-lg border p-3"
   477→                      >
   478→                        <div className="flex items-start justify-between gap-2 flex-wrap">
   479→                          <div className="flex-1 min-w-[200px]">
   480→                            <div className="flex items-center gap-2 mb-1">
   481→                              <Badge variant="outline" className={`text-[10px] ${ACTION_COLOR[a.action.actionType] ?? ""}`}>
   482→                                {a.action.actionType}
   483→                              </Badge>
   484→                              <span className="font-medium text-sm">{a.user.displayName}</span>
   485→                              <span className="text-[10px] text-muted-foreground">@{a.user.username}</span>
   486→                            </div>
   487→                            <p className="text-xs text-muted-foreground">
   488→                              Original reason: <span className="text-foreground">{a.action.reason}</span>
   489→                            </p>
   490→                            <p className="text-sm mt-1">{a.reason}</p>
   491→                            <p className="text-[10px] text-muted-foreground mt-1">
   492→                              Filed {new Date(a.createdAt).toLocaleString()}
   493→                            </p>
   494→                          </div>
   495→                          <div className="flex gap-2">
   496→                            <Button
   497→                              size="sm"
   498→                              variant="outline"
   499→                              className="border-emerald-300 text-emerald-700 hover:bg-emerald-50"
   500→                              disabled={busyId === a.id}
   501→                              onClick={() => reviewAppeal(a.id, "APPROVED")}
   502→                            >
   503→                              <Check className="w-3 h-3 mr-1" /> Approve
   504→                            </Button>
   505→                            <Button
   506→                              size="sm"
   507→                              variant="outline"
   508→                              className="border-rose-300 text-rose-700 hover:bg-rose-50"
   509→                              disabled={busyId === a.id}
   510→                              onClick={() => reviewAppeal(a.id, "DENIED")}
   511→                            >
   512→                              <X className="w-3 h-3 mr-1" /> Deny
   513→                            </Button>
   514→                          </div>
   515→                        </div>
   516→                      </motion.div>
   517→                    ))}
   518→                  </div>
   519→                )}
   520→              </CardContent>
   521→            </Card>
   522→          </TabsContent>
   523→        </Tabs>
   524→      )}
   525→
   526→      {/* Inline action target launcher (used by Take Action dialog) */}
   527→      {actionTarget && (
   528→        <TakeActionDialog
   529→          open
   530→          target={actionTarget}
   531→          onConfirm={takeAction}
   532→          busy={busyId === actionTarget.userId}
   533→          onClose={() => setActionTarget(null)}
   534→        />
   535→      )}
   536→    </div>
   537→  );
   538→}

   559→// ============================================================
   560→// Take Action Dialog
   561→// ============================================================
   562→function TakeActionDialog({
   563→  open, target, onConfirm, busy, onClose,
   564→}: {
   565→  open?: boolean;
   566→  target: { userId: string; displayName: string };
   567→  onConfirm: (p: { actionType: string; userId: string; reason: string; durationMinutes: number }) => void;
   568→  busy: boolean;
   569→  onClose?: () => void;
   570→}) {
   571→  const [actionType, setActionType] = useState("WARNING");
   572→  const [reason, setReason] = useState("");
   573→  const [durationMinutes, setDurationMinutes] = useState(60);
   574→  const [internalOpen, setInternalOpen] = useState(false);
   575→
   576→  const isOpen = open ?? internalOpen;
   577→
   578→  function handleSubmit() {
   579→    if (!reason.trim()) {
   580→      toast.error("Reason is required");
   581→      return;
   582→    }
   583→    onConfirm({
   584→      actionType,
   585→      userId: target.userId,
   586→      reason: reason.trim(),
   587→      durationMinutes: actionType === "MUTE" || actionType === "SUSPEND" ? durationMinutes : 0,
   588→    });
   589→    setReason("");
   590→    setActionType("WARNING");
   591→    setDurationMinutes(60);
   592→    setInternalOpen(false);
   593→    onClose?.();
   594→  }
   595→
   596→  return (
   597→    <Dialog open={isOpen} onOpenChange={(v) => {
   598→      setInternalOpen(v);
   599→      if (!v) onClose?.();
   600→    }}>
   601→      <DialogTrigger asChild>
   602→        {open ? undefined : (
   603→          <Button size="sm" variant="destructive">
   604→            <Gavel className="w-3 h-3 mr-1" /> Take Action
   605→          </Button>
   606→        )}
   607→      </DialogTrigger>
   608→      <DialogContent>
   609→        <DialogHeader>
   610→          <DialogTitle>Take Moderation Action</DialogTitle>
   611→          <DialogDescription>
   612→            Acting on <span className="font-semibold">{target.displayName}</span>.
   613→          </DialogDescription>
   614→        </DialogHeader>
   615→        <div className="space-y-3">
   616→          <div>
   617→            <Label>Action Type</Label>
   618→            <Select value={actionType} onValueChange={setActionType}>
   619→              <SelectTrigger>
   620→                <SelectValue />
   621→              </SelectTrigger>
   622→              <SelectContent>
   623→                {["WARNING", "MUTE", "SUSPEND", "BAN"].map((a) => {
   624→                  const Icon = ACTION_ICON[a] ?? Bell;
   625→                  return (
   626→                    <SelectItem key={a} value={a}>
   627→                      <span className="flex items-center gap-1">
   628→                        <Icon className="w-3 h-3" /> {a}
   629→                      </span>
   630→                    </SelectItem>
   631→                  );
   632→                })}
   633→              </SelectContent>
   634→            </Select>
   635→          </div>
   636→          {(actionType === "MUTE" || actionType === "SUSPEND") && (
   637→            <div>
   638→              <Label>Duration (minutes)</Label>
   639→              <Input
   640→                type="number"
   641→                min={1}
   642→                value={durationMinutes}
   643→                onChange={(e) => setDurationMinutes(parseInt(e.target.value || "0", 10))}
   644→              />
   645→              <p className="text-[10px] text-muted-foreground mt-1">
   646→                Set 0 for permanent (not recommended for first offenses).
   647→              </p>
   648→            </div>
   649→          )}
   650→          <div>
   651→            <Label>Reason</Label>
   652→            <Textarea
   653→              value={reason}
   654→              onChange={(e) => setReason(e.target.value)}
   655→              placeholder="Explain why this action is being taken..."
   656→              rows={3}
   657→            />
   658→          </div>
   659→        </div>
   660→        <DialogFooter>
   661→          <Button variant="outline" onClick={() => { setInternalOpen(false); onClose?.(); }} disabled={busy}>
   662→            Cancel
   663→          </Button>
   664→          <Button variant="destructive" onClick={handleSubmit} disabled={busy}>
   665→            {busy ? <Loader2 className="w-4 h-4 mr-1 animate-spin" /> : <Gavel className="w-4 h-4 mr-1" />}
   666→            Apply Action
   667→          </Button>
   668→        </DialogFooter>
   669→      </DialogContent>
   670→    </Dialog>
   671→  );
   672→}
