     3→/**
     4→ * MANAHAD — Game World View
     5→ *
     6→ * A 2D top-down canvas world where players can:
     7→ *   - Walk around (WASD / arrow keys / touch joystick)
     8→ *   - See other players moving in realtime (via WebSocket)
     9→ *   - Chat with nearby players (chat panel)
    10→ *   - Walk into portals to change areas
    11→ *   - Talk to NPCs (opens dialogue)
    12→ *   - Use emotes
    13→ *
    14→ * The world is rendered on a canvas with requestAnimationFrame.
    15→ * Player position is synced to the server at ~30Hz.
    16→ */
    17→

    19→import { motion, AnimatePresence } from "framer-motion";
    20→import { Button } from "@/components/ui/button";
    21→import { Input } from "@/components/ui/input";

    25→import { useMultiplayerStore } from "@/stores/multiplayer-store";
    26→import { renderAvatar } from "./avatar-renderer";
    27→import { WORLD_AREAS, NPCS, getAreaBySlug, getNPCBySlug } from "@/lib/game/world";
    28→import { toast } from "sonner";

    33→interface ActiveEmote {
    34→  userId: string;
    35→  emoji: string;
    36→  startFrame: number;

    38→
    39→export function WorldView() {
    40→  const user = useAppStore((s) => s.user);
    41→  const setView = useAppStore((s) => s.setView);
    42→  const updateUser = useAppStore((s) => s.updateUser);
    43→  const dyslexiaFont = useAppStore((s) => s.dyslexiaFont);
    44→
    45→  const canvasRef = useRef<HTMLCanvasElement>(null);
    46→  const containerRef = useRef<HTMLDivElement>(null);
    47→
    48→  const {
    49→    connect, disconnect, connected, players, messages, currentArea,
    50→    moveTo, changeArea, sendChat, sendEmote,
    51→  } = useMultiplayerStore();
    52→
    53→  // Local player position (high-frequency, synced to server)
    54→  const playerPos = useRef({ x: 500, y: 400, direction: "down" as const, isMoving: false });
    55→  const keysPressed = useRef<Set<string>>(new Set());
    56→  const animFrame = useRef(0);
    57→  const activeEmotes = useRef<ActiveEmote[]>([]);
    58→  const [chatInput, setChatInput] = useState("");
    59→  const [showChat, setShowChat] = useState(true);
    60→  const [showMobileControls, setShowMobileControls] = useState(false);
    61→  const [activeNPC, setActiveNPC] = useState<string | null>(null);
    62→
    63→  const area = getAreaBySlug(currentArea) ?? WORLD_AREAS[0];
    64→
    65→  // === Connect to multiplayer ===
    66→  useEffect(() => {
    67→    if (!user) return;
    68→    connect({
    69→      userId: user.id,
    70→      username: user.username,
    71→      displayName: user.displayName,
    72→      avatarConfig: user.avatarConfig,
    73→      area: "town",
    74→      level: user.level,
    75→    });
    76→    return () => disconnect();
    77→  }, [user?.id]);
    78→
    79→  // === Listen for emote events ===
    80→  useEffect(() => {
    81→    function onEmote(e: Event) {
    82→      const detail = (e as CustomEvent).detail;
    83→      activeEmotes.current.push({
    84→        userId: detail.userId,
    85→        emoji: detail.emoteSlug === "emote-wave" ? "👋" :
    86→               detail.emoteSlug === "emote-dance" ? "💃" :
    87→               detail.emoteSlug === "emote-cheer" ? "🎉" :
    88→               detail.emoteSlug === "emote-thinking" ? "🤔" : "✨",
    89→        startFrame: animFrame.current,
    90→      });
    91→    }
    92→    window.addEventListener("mv:emote", onEmote);
    93→    return () => window.removeEventListener("mv:emote", onEmote);
    94→  }, []);
    95→
    96→  // === Keyboard input ===
    97→  useEffect(() => {
    98→    function onKey(e: KeyboardEvent) {
    99→      const key = e.key.toLowerCase();
   100→      if (["w", "a", "s", "d", "arrowup", "arrowdown", "arrowleft", "arrowright"].includes(key)) {
   101→        e.preventDefault();
   102→        keysPressed.current.add(key);
   103→      }
   104→      // Quick emotes
   105→      if (key === "e") sendEmote("emote-wave");
   106→      if (key === "r") sendEmote("emote-dance");
   107→      if (key === "t") sendEmote("emote-thinking");
   108→    }
   109→    function onKeyUp(e: KeyboardEvent) {
   110→      keysPressed.current.delete(e.key.toLowerCase());
   111→    }
   112→    window.addEventListener("keydown", onKey);
   113→    window.addEventListener("keyup", onKeyUp);
   114→    return () => {
   115→      window.removeEventListener("keydown", onKey);
   116→      window.removeEventListener("keyup", onKeyUp);
   117→    };
   118→  }, [sendEmote]);
   119→
   120→  // === Touch joystick ===
   121→  const joystickRef = useRef<{ active: boolean; dx: number; dy: number }>({ active: false, dx: 0, dy: 0 });
   122→  const [joystickOffset, setJoystickOffset] = useState({ dx: 0, dy: 0 });
   123→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   124→    const rect = (e.target as HTMLElement).getBoundingClientRect();
   125→    const touch = e.touches[0];
   126→    joystickRef.current = {
   127→      active: true,
   128→      dx: touch.clientX - (rect.left + rect.width / 2),
   129→      dy: touch.clientY - (rect.top + rect.height / 2),
   130→    };
   131→    setJoystickOffset({ dx: joystickRef.current.dx, dy: joystickRef.current.dy });
   132→  }, []);
   133→  const onTouchMove = useCallback((e: React.TouchEvent) => {
   134→    if (!joystickRef.current.active) return;
   135→    const rect = (e.target as HTMLElement).getBoundingClientRect();
   136→    const touch = e.touches[0];
   137→    joystickRef.current.dx = touch.clientX - (rect.left + rect.width / 2);
   138→    joystickRef.current.dy = touch.clientY - (rect.top + rect.height / 2);
   139→    setJoystickOffset({ dx: joystickRef.current.dx, dy: joystickRef.current.dy });
   140→  }, []);
   141→  const onTouchEnd = useCallback(() => {
   142→    joystickRef.current = { active: false, dx: 0, dy: 0 };
   143→    setJoystickOffset({ dx: 0, dy: 0 });
   144→  }, []);
   145→
   146→  // === Main game loop ===
   147→  useEffect(() => {
   148→    const canvas = canvasRef.current;
   149→    if (!canvas) return;
   150→    const ctx = canvas.getContext("2d");
   151→    if (!ctx) return;
   152→
   153→    let rafId: number;
   154→
   155→    function resize() {
   156→      if (!canvas || !containerRef.current) return;
   157→      const dpr = window.devicePixelRatio || 1;
   158→      const rect = containerRef.current.getBoundingClientRect();
   159→      canvas.width = rect.width * dpr;
   160→      canvas.height = rect.height * dpr;
   161→      canvas.style.width = `${rect.width}px`;
   162→      canvas.style.height = `${rect.height}px`;
   163→      ctx!.setTransform(dpr, 0, 0, dpr, 0, 0);
   164→    }
   165→    resize();
   166→    window.addEventListener("resize", resize);
   167→
   168→    let lastServerSync = 0;
   169→
   170→    function tick() {
   171→      animFrame.current++;
   172→      const rect = canvas!.getBoundingClientRect();
   173→      const w = rect.width;
   174→      const h = rect.height;
   175→
   176→      // === Update local player position from input ===
   177→      let dx = 0, dy = 0;
   178→      const keys = keysPressed.current;
   179→      if (keys.has("w") || keys.has("arrowup")) dy -= 1;
   180→      if (keys.has("s") || keys.has("arrowdown")) dy += 1;
   181→      if (keys.has("a") || keys.has("arrowleft")) dx -= 1;
   182→      if (keys.has("d") || keys.has("arrowright")) dx += 1;
   183→
   184→      // Joystick input
   185→      if (joystickRef.current.active) {
   186→        const j = joystickRef.current;
   187→        const mag = Math.sqrt(j.dx * j.dx + j.dy * j.dy);
   188→        if (mag > 10) {
   189→          dx = j.dx / mag;
   190→          dy = j.dy / mag;
   191→        }
   192→      }
   193→
   194→      const speed = 3.5;
   195→      const moving = (dx !== 0 || dy !== 0);
   196→      if (moving) {
   197→        const norm = Math.sqrt(dx * dx + dy * dy) || 1;
   198→        const ndx = dx / norm;
   199→        const ndy = dy / norm;
   200→        playerPos.current.x = Math.max(20, Math.min(area.width - 20, playerPos.current.x + ndx * speed));
   201→        playerPos.current.y = Math.max(60, Math.min(area.height - 20, playerPos.current.y + ndy * speed));
   202→        // Set direction (dominant axis)
   203→        if (Math.abs(ndx) > Math.abs(ndy)) {
   204→          playerPos.current.direction = ndx > 0 ? "right" : "left";
   205→        } else {
   206→          playerPos.current.direction = ndy > 0 ? "down" : "up";
   207→        }
   208→        playerPos.current.isMoving = true;
   209→      } else {
   210→        playerPos.current.isMoving = false;
   211→      }
   212→
   213→      // === Sync to server (~30Hz) ===
   214→      const now = Date.now();
   215→      if (now - lastServerSync > 33 && moving) {
   216→        moveTo(playerPos.current.x, playerPos.current.y, playerPos.current.direction);
   217→        lastServerSync = now;
   218→      }
   219→
   220→      // === Camera follows player ===
   221→      const camX = playerPos.current.x - w / 2;
   222→      const camY = playerPos.current.y - h / 2;
   223→
   224→      // === Clear ===
   225→      ctx!.fillStyle = area.bgColor;
   226→      ctx!.fillRect(0, 0, w, h);
   227→
   228→      // === Camera transform ===
   229→      ctx!.save();
   230→      ctx!.translate(-camX, -camY);
   231→
   232→      // === Background pattern (subtle grid) ===
   233→      ctx!.strokeStyle = "rgba(255,255,255,0.08)";
   234→      ctx!.lineWidth = 1;
   235→      const gridSize = 60;
   236→      const startX = Math.floor(camX / gridSize) * gridSize;
   237→      const startY = Math.floor(camY / gridSize) * gridSize;
   238→      for (let x = startX; x < camX + w; x += gridSize) {
   239→        ctx!.beginPath();
   240→        ctx!.moveTo(x, camY);
   241→        ctx!.lineTo(x, camY + h);
   242→        ctx!.stroke();
   243→      }
   244→      for (let y = startY; y < camY + h; y += gridSize) {
   245→        ctx!.beginPath();
   246→        ctx!.moveTo(camX, y);
   247→        ctx!.lineTo(camX + w, y);
   248→        ctx!.stroke();
   249→      }
   250→
   251→      // === Area boundary ===
   252→      ctx!.strokeStyle = "rgba(0,0,0,0.2)";
   253→      ctx!.lineWidth = 4;
   254→      ctx!.strokeRect(0, 0, area.width, area.height);
   255→
   256→      // === Decorations ===
   257→      for (const dec of area.decorations) {
   258→        ctx!.fillStyle = dec.color;
   259→        ctx!.strokeStyle = "rgba(0,0,0,0.15)";
   260→        ctx!.lineWidth = 2;
   261→        if (dec.type === "tree") {
   262→          // Trunk
   263→          ctx!.fillStyle = "#5a3a1a";
   264→          ctx!.fillRect(dec.x + dec.w / 2 - 5, dec.y + dec.h / 2, 10, dec.h / 2);
   265→          // Leaves
   266→          ctx!.fillStyle = dec.color;
   267→          ctx!.beginPath();
   268→          ctx!.arc(dec.x + dec.w / 2, dec.y + dec.h / 2, dec.w / 2, 0, Math.PI * 2);
   269→          ctx!.fill();
   270→          ctx!.stroke();
   271→        } else if (dec.type === "pond") {
   272→          ctx!.fillStyle = dec.color;
   273→          ctx!.beginPath();
   274→          ctx!.ellipse(dec.x + dec.w / 2, dec.y + dec.h / 2, dec.w / 2, dec.h / 2, 0, 0, Math.PI * 2);
   275→          ctx!.fill();
   276→          // Ripple
   277→          ctx!.strokeStyle = "rgba(255,255,255,0.4)";
   278→          ctx!.beginPath();
   279→          ctx!.ellipse(dec.x + dec.w / 2, dec.y + dec.h / 2, dec.w / 3, dec.h / 3, 0, 0, Math.PI * 2);
   280→          ctx!.stroke();
   281→        } else if (dec.type === "building") {
   282→          ctx!.fillRect(dec.x, dec.y, dec.w, dec.h);
   283→          ctx!.strokeRect(dec.x, dec.y, dec.w, dec.h);
   284→          // Roof
   285→          ctx!.fillStyle = "rgba(0,0,0,0.2)";
   286→          ctx!.beginPath();
   287→          ctx!.moveTo(dec.x, dec.y);
   288→          ctx!.lineTo(dec.x + dec.w / 2, dec.y - 15);
   289→          ctx!.lineTo(dec.x + dec.w, dec.y);
   290→          ctx!.closePath();
   291→          ctx!.fill();
   292→        } else if (dec.type === "flower") {
   293→          ctx!.beginPath();
   294→          ctx!.arc(dec.x + 5, dec.y + 5, 4, 0, Math.PI * 2);
   295→          ctx!.arc(dec.x + 15, dec.y + 5, 4, 0, Math.PI * 2);
   296→          ctx!.arc(dec.x + 10, dec.y, 4, 0, Math.PI * 2);
   297→          ctx!.arc(dec.x + 10, dec.y + 10, 4, 0, Math.PI * 2);
   298→          ctx!.arc(dec.x + 10, dec.y + 5, 3, 0, Math.PI * 2);
   299→          ctx!.fill();
   300→        } else if (dec.type === "machine" || dec.type === "stall") {
   301→          ctx!.fillRect(dec.x, dec.y, dec.w, dec.h);
   302→          ctx!.strokeRect(dec.x, dec.y, dec.w, dec.h);
   303→          // Glow
   304→          ctx!.fillStyle = "rgba(255,255,255,0.3)";
   305→          ctx!.fillRect(dec.x + 5, dec.y + 5, dec.w - 10, 8);
   306→        } else if (dec.type === "neon-sign" || dec.type === "awning" || dec.type === "rug" || dec.type === "path") {
   307→          ctx!.fillRect(dec.x, dec.y, dec.w, dec.h);
   308→        } else if (dec.type === "door") {
   309→          ctx!.fillRect(dec.x, dec.y, dec.w, dec.h);
   310→          ctx!.fillStyle = "#fbbf24";
   311→          ctx!.beginPath();
   312→          ctx!.arc(dec.x + dec.w - 8, dec.y + dec.h / 2, 2, 0, Math.PI * 2);
   313→          ctx!.fill();
   314→        } else if (dec.type === "blackboard") {
   315→          ctx!.fillRect(dec.x, dec.y, dec.w, dec.h);
   316→          // Chalk text
   317→          ctx!.fillStyle = "#ffffff";
   318→          ctx!.font = "12px var(--font-geist-sans), sans-serif";
   319→          ctx!.textAlign = "center";
   320→          ctx!.fillText("MATH = MAGIC ✨", dec.x + dec.w / 2, dec.y + dec.h / 2 + 4);
   321→        } else if (dec.type === "desk") {
   322→          ctx!.fillRect(dec.x, dec.y, dec.w, dec.h);
   323→          ctx!.strokeRect(dec.x, dec.y, dec.w, dec.h);
   324→        } else if (dec.type === "bench") {
   325→          ctx!.fillRect(dec.x, dec.y, dec.w, dec.h);
   326→        } else if (dec.type === "fountain") {
   327→          ctx!.fillStyle = dec.color;
   328→          ctx!.beginPath();
   329→          ctx!.ellipse(dec.x + dec.w / 2, dec.y + dec.h / 2, dec.w / 2, dec.h / 2, 0, 0, Math.PI * 2);
   330→          ctx!.fill();
   331→          ctx!.fillStyle = "rgba(255,255,255,0.4)";
   332→          ctx!.beginPath();
   333→          ctx!.ellipse(dec.x + dec.w / 2, dec.y + dec.h / 2, dec.w / 3, dec.h / 3, 0, 0, Math.PI * 2);
   334→          ctx!.fill();
   335→        } else {
   336→          ctx!.fillRect(dec.x, dec.y, dec.w, dec.h);
   337→        }
   338→      }
   339→
   340→      // === Portals ===
   341→      for (const portal of area.portals) {
   342→        const pulse = Math.sin(animFrame.current * 0.06) * 0.2 + 0.8;
   343→        ctx!.fillStyle = `rgba(168, 85, 247, ${0.3 * pulse})`;
   344→        ctx!.beginPath();
   345→        ctx!.arc(portal.x, portal.y, 30, 0, Math.PI * 2);
   346→        ctx!.fill();
   347→        ctx!.strokeStyle = `rgba(168, 85, 247, ${pulse})`;
   348→        ctx!.lineWidth = 3;
   349→        ctx!.stroke();
   350→        // Label
   351→        ctx!.fillStyle = "#1f2937";
   352→        ctx!.font = "600 11px var(--font-geist-sans), sans-serif";
   353→        ctx!.textAlign = "center";
   354→        ctx!.fillText(portal.label, portal.x, portal.y - 35);
   355→
   356→        // Check if player is on portal — trigger transition directly (debounced)
   357→        const dist = Math.sqrt((portal.x - playerPos.current.x) ** 2 + (portal.y - playerPos.current.y) ** 2);
   358→        if (dist < 35) {
   359→          const now = Date.now();
   360→          if (now - lastTransitionAt.current > 1500) {
   361→            lastTransitionAt.current = now;
   362→            // Defer to next tick to avoid setState during render
   363→            setTimeout(() => doAreaTransition(portal.target), 0);
   364→          }
   365→        }
   366→      }
   367→
   368→      // === NPCs ===
   369→      for (const npcDef of area.npcs) {
   370→        const npc = getNPCBySlug(npcDef.slug);
   371→        if (!npc) continue;
   372→        // Render as a colored circle with emoji
   373→        ctx!.fillStyle = npc.color;
   374→        ctx!.beginPath();
   375→        ctx!.arc(npcDef.x, npcDef.y, 18, 0, Math.PI * 2);
   376→        ctx!.fill();
   377→        ctx!.font = "24px sans-serif";
   378→        ctx!.textAlign = "center";
   379→        ctx!.textBaseline = "middle";
   380→        ctx!.fillText(npc.emoji, npcDef.x, npcDef.y);
   381→        // Name
   382→        ctx!.font = "600 10px var(--font-geist-sans), sans-serif";
   383→        ctx!.fillStyle = "#1f2937";
   384→        ctx!.fillText(npc.name, npcDef.x, npcDef.y + 30);
   385→
   386→        // "Talk" indicator if player is close
   387→        const dist = Math.sqrt((npcDef.x - playerPos.current.x) ** 2 + (npcDef.y - playerPos.current.y) ** 2);
   388→        if (dist < 50) {
   389→          ctx!.fillStyle = "#10b981";
   390→          ctx!.font = "700 11px var(--font-geist-sans), sans-serif";
   391→          const bob = Math.sin(animFrame.current * 0.1) * 2;
   392→          ctx!.fillText("Press T to talk", npcDef.x, npcDef.y - 30 + bob);
   393→        }
   394→      }
   395→
   396→      // === Other players (remote) ===
   397→      for (const p of players) {
   398→        if (p.userId === user?.id) continue;
   399→        const emote = activeEmotes.current.find((e) => e.userId === p.userId);
   400→        renderAvatar({
   401→          ctx: ctx!,
   402→          x: p.x,
   403→          y: p.y,
   404→          size: 40,
   405→          direction: p.direction,
   406→          isMoving: p.isMoving,
   407→          animFrame: animFrame.current,
   408→          displayName: p.displayName,
   409→          showName: true,
   410→          config: p.avatarConfig,
   411→          emote: emote ? emote.emoji : null,
   412→        });
   413→      }
   414→
   415→      // === Local player ===
   416→      renderAvatar({
   417→        ctx: ctx!,
   418→        x: playerPos.current.x,
   419→        y: playerPos.current.y,
   420→        size: 40,
   421→        direction: playerPos.current.direction,
   422→        isMoving: playerPos.current.isMoving,
   423→        animFrame: animFrame.current,
   424→        displayName: user?.displayName,
   425→        showName: true,
   426→        isSelected: true,
   427→        config: user?.avatarConfig ?? ({} as any),
   428→      });
   429→
   430→      ctx!.restore();
   431→
   432→      rafId = requestAnimationFrame(tick);
   433→    }
   434→
   435→    rafId = requestAnimationFrame(tick);
   436→
   437→    return () => {
   438→      cancelAnimationFrame(rafId);
   439→      window.removeEventListener("resize", resize);
   440→    };
   441→  }, [area, user?.id, players]);
   442→
   443→  // === Handle "T" to talk to NPC ===
   444→  useEffect(() => {
   445→    function onKey(e: KeyboardEvent) {
   446→      if (e.key.toLowerCase() === "t" && !activeNPC) {
   447→        // Find nearby NPC
   448→        for (const npcDef of area.npcs) {
   449→          const dist = Math.sqrt((npcDef.x - playerPos.current.x) ** 2 + (npcDef.y - playerPos.current.y) ** 2);
   450→          if (dist < 50) {
   451→            setActiveNPC(npcDef.slug);
   452→            return;
   453→          }
   454→        }
   455→      }
   456→      if (e.key === "Escape") {
   457→        setActiveNPC(null);
   458→      }
   459→    }
   460→    window.addEventListener("keydown", onKey);
   461→    return () => window.removeEventListener("keydown", onKey);
   462→  }, [area, activeNPC]);
   463→
   464→  // === Handle area transition (called directly from game loop, no effect needed) ===
   465→  const doAreaTransition = useCallback((targetSlug: string) => {
   466→    const targetArea = getAreaBySlug(targetSlug);
   467→    if (!targetArea) return;
   468→    toast.info(`Entering ${targetArea.name}...`);
   469→    changeArea(targetArea.slug, targetArea.spawnPoint.x, targetArea.spawnPoint.y);
   470→    // Update local position via ref (mutation is intentional — playerPos is a high-frequency ref, not state)
   471→    playerPos.current.x = targetArea.spawnPoint.x;
   472→    playerPos.current.y = targetArea.spawnPoint.y;
   473→  }, [changeArea]);
   474→
   475→  // Track last portal transition to debounce
   476→  const lastTransitionAt = useRef(0);
   477→
   478→  function sendChatMessage() {
   479→    if (!chatInput.trim()) return;
   480→    // Brain Energy cost check
   481→    if ((user?.brainEnergy ?? 0) < 1) {
   482→      toast.error("Out of Brain Energy! Answer some math questions to chat.");
   483→      return;
   484→    }
   485→    sendChat(chatInput);
   486→    setChatInput("");
   487→    // Deduct brain energy locally
   488→    if (user) updateUser({ brainEnergy: Math.max(0, user.brainEnergy - 1) });
   489→  }
   490→
   491→  if (!user) return null;
   492→
   493→  return (
   494→    <div className={`relative w-full h-full ${dyslexiaFont ? "dyslexia-font" : ""}`}>
   495→      {/* Top bar: area info, brain energy, players count */}
   496→      <div className="absolute top-0 left-0 right-0 z-20 p-3 flex items-center justify-between gap-2 pointer-events-none">
   497→        <div className="flex items-center gap-2 pointer-events-auto">
