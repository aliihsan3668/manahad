   155→   155→      const res = await fetch("/api/moderator/action", {
   156→   156→        method: "POST",

   166→   166→      toast.error(e instanceof Error ? e.message : "Action failed");
   167→   167→    } finally {
   168→   168→      setBusyId(null);

   442→   442→                                  setBusyId(null);
   443→   443→                                }
   444→   444→                              }}
   445→   445→                            >
   446→   446→                              {busyId === a.id ? <Loader2 className="w-3 h-3 mr-1 animate-spin" /> : <Check className="w-3 h-3 mr-1" />}
   447→   447→                              Lift
   448→   448→                            </Button>
   449→   449→                          </TableCell>
   450→   450→                        </TableRow>
   451→   451→                      ))}
   452→   452→                    </TableBody>
   453→   453→                  </Table>
   454→   454→                )}
   455→   455→              </CardContent>
   456→   456→            </Card>
   457→   457→          </TabsContent>
   458→   458→
   459→   459→          {/* ============== APPEALS ============== */}
   460→   460→          <TabsContent value="appeals">
   461→   461→            <Card>
   462→   462→              <CardHeader>
   463→   463→                <CardTitle className="text-lg">Open Appeals</CardTitle>
   464→   464→                <CardDescription>Users contesting moderation actions.</CardDescription>
   465→   465→              </CardHeader>
   466→   466→              <CardContent>
   467→   467→                {data.openAppeals.length === 0 ? (
   468→   468→                  <EmptyState text="No open appeals." />
   469→   469→                ) : (
   470→   470→                  <div className="space-y-3">
   471→   471→                    {data.openAppeals.map((a) => (
   472→   472→                      <motion.div
   473→   473→                        key={a.id}
   474→   474→                        initial={{ opacity: 0, y: 4 }}
   475→   475→                        animate={{ opacity: 1, y: 0 }}
   476→   476→                        className="rounded-lg border p-3"
   477→   477→                      >
   478→   478→                        <div className="flex items-start justify-between gap-2 flex-wrap">
   479→   479→                          <div className="flex-1 min-w-[200px]">
   480→   480→                            <div className="flex items-center gap-2 mb-1">
   481→   481→                              <Badge variant="outline" className={`text-[10px] ${ACTION_COLOR[a.action.actionType] ?? ""}`}>
   482→   482→                                {a.action.actionType}
   483→   483→                              </Badge>
   484→   484→                              <span className="font-medium text-sm">{a.user.displayName}</span>
   485→   485→                              <span className="text-[10px] text-muted-foreground">@{a.user.username}</span>

   489→   489→                            </p>
   490→   490→                            <p className="text-sm mt-1">{a.reason}</p>
   491→   491→                            <p className="text-[10px] text-muted-foreground mt-1">
   492→   492→                              Filed {new Date(a.createdAt).toLocaleString()}
   493→   493→                            </p>

   501→   501→                              onClick={() => reviewAppeal(a.id, "APPROVED")}
   502→   502→                            >
   503→   503→                              <Check className="w-3 h-3 mr-1" /> Approve

   505→   505→                            <Button
   506→   506→                              size="sm"
   507→   507→                              variant="outline"
   508→   508→                              className="border-rose-300 text-rose-700 hover:bg-rose-50"
   509→   509→                              disabled={busyId === a.id}
   510→   510→                              onClick={() => reviewAppeal(a.id, "DENIED")}
   511→   511→                            >
   512→   512→                              <X className="w-3 h-3 mr-1" /> Deny
   513→   513→                            </Button>
   514→   514→                          </div>
   515→   515→                        </div>
   516→   516→                      </motion.div>

   518→   518→                  </div>
   519→   519→                )}
   520→   520→              </CardContent>
   521→   521→            </Card>
   522→   522→          </TabsContent>
   523→   523→        </Tabs>
   524→   524→      )}
   525→   525→
   526→   526→      {/* Inline action target launcher (used by Take Action dialog) */}

   534→   534→        />
   535→   535→      )}
   536→   536→    </div>
   537→   537→  );
   538→   538→}

   560→   560→// Take Action Dialog
   561→   561→// ============================================================
   562→   562→function TakeActionDialog({
   563→   563→  open, target, onConfirm, busy, onClose,
   564→   564→}: {
   565→   565→  open?: boolean;
   566→   566→  target: { userId: string; displayName: string };
   567→   567→  onConfirm: (p: { actionType: string; userId: string; reason: string; durationMinutes: number }) => void;
   568→   568→  busy: boolean;
   569→   569→  onClose?: () => void;
   570→   570→}) {
   571→   571→  const [actionType, setActionType] = useState("WARNING");
   572→   572→  const [reason, setReason] = useState("");
   573→   573→  const [durationMinutes, setDurationMinutes] = useState(60);
   574→   574→  const [internalOpen, setInternalOpen] = useState(false);
   575→   575→
   576→   576→  const isOpen = open ?? internalOpen;
   577→   577→
   578→   578→  function handleSubmit() {
   579→   579→    if (!reason.trim()) {
   580→   580→      toast.error("Reason is required");
   581→   581→      return;
   582→   582→    }
   583→   583→    onConfirm({
   584→   584→      actionType,
   585→   585→      userId: target.userId,
   586→   586→      reason: reason.trim(),
   587→   587→      durationMinutes: actionType === "MUTE" || actionType === "SUSPEND" ? durationMinutes : 0,
   588→   588→    });
   589→   589→    setReason("");
   590→   590→    setActionType("WARNING");
   591→   591→    setDurationMinutes(60);
   592→   592→    setInternalOpen(false);
   593→   593→    onClose?.();
   594→   594→  }
   595→   595→
   596→   596→  return (
   597→   597→    <Dialog open={isOpen} onOpenChange={(v) => {
   598→   598→      setInternalOpen(v);
   599→   599→      if (!v) onClose?.();
   600→   600→    }}>
   601→   601→      <DialogTrigger asChild>
   602→   602→        {open ? undefined : (
   603→   603→          <Button size="sm" variant="destructive">
   604→   604→            <Gavel className="w-3 h-3 mr-1" /> Take Action
   605→   605→          </Button>
   606→   606→        )}
   607→   607→      </DialogTrigger>
   608→   608→      <DialogContent>
   609→   609→        <DialogHeader>
   610→   610→          <DialogTitle>Take Moderation Action</DialogTitle>
   611→   611→          <DialogDescription>
   612→   612→            Acting on <span className="font-semibold">{target.displayName}</span>.
   613→   613→          </DialogDescription>
   614→   614→        </DialogHeader>
   615→   615→        <div className="space-y-3">
   616→   616→          <div>
   617→   617→            <Label>Action Type</Label>
   618→   618→            <Select value={actionType} onValueChange={setActionType}>
   619→   619→              <SelectTrigger>
   620→   620→                <SelectValue />
   621→   621→              </SelectTrigger>
   622→   622→              <SelectContent>
   623→   623→                {["WARNING", "MUTE", "SUSPEND", "BAN"].map((a) => {
   624→   624→                  const Icon = ACTION_ICON[a] ?? Bell;
   625→   625→                  return (
   626→   626→                    <SelectItem key={a} value={a}>
   627→   627→                      <span className="flex items-center gap-1">
   628→   628→                        <Icon className="w-3 h-3" /> {a}
   629→   629→                      </span>
   630→   630→                    </SelectItem>
   631→   631→                  );
   632→   632→                })}
   633→   633→              </SelectContent>
   634→   634→            </Select>
   635→   635→          </div>
   636→   636→          {(actionType === "MUTE" || actionType === "SUSPEND") && (
   637→   637→            <div>
   638→   638→              <Label>Duration (minutes)</Label>
   639→   639→              <Input
   640→   640→                type="number"
   641→   641→                min={1}
   642→   642→                value={durationMinutes}
   643→   643→                onChange={(e) => setDurationMinutes(parseInt(e.target.value || "0", 10))}
   644→   644→              />
   645→   645→              <p className="text-[10px] text-muted-foreground mt-1">
   646→   646→                Set 0 for permanent (not recommended for first offenses).
   647→   647→              </p>
   648→   648→            </div>
   649→   649→          )}
   650→   650→          <div>
   651→   651→            <Label>Reason</Label>
   652→   652→            <Textarea
   653→   653→              value={reason}
   654→   654→              onChange={(e) => setReason(e.target.value)}
   655→   655→              placeholder="Explain why this action is being taken..."
   656→   656→              rows={3}
   657→   657→            />
   658→   658→          </div>
   659→   659→        </div>
   660→   660→        <DialogFooter>
   661→   661→          <Button variant="outline" onClick={() => { setInternalOpen(false); onClose?.(); }} disabled={busy}>
   662→   662→            Cancel
   663→   663→          </Button>
   664→   664→          <Button variant="destructive" onClick={handleSubmit} disabled={busy}>
   665→   665→            {busy ? <Loader2 className="w-4 h-4 mr-1 animate-spin" /> : <Gavel className="w-4 h-4 mr-1" />}
   666→   666→            Apply Action
   667→   667→          </Button>
   668→   668→        </DialogFooter>
   669→   669→      </DialogContent>
   670→   670→    </Dialog>
   671→   671→  );
   672→   672→}
